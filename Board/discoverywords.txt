// Discovery board CLI bindings
// Run as prebuild step: python parsewords.py <path>/discoverywords.txt
// Generates: help.c  wordlist.c  wordlist.txt
// Include clibindings.txt and floatwords.txt for full CLI support.

Constants
	gnetif			gnetif			// ( - a ) LwIP netif structure for Ethernet
End

Words // network
	show-eth		show_ethernet	// show Ethernet link status, speed, duplex and PHY info
	show-ip			show_ip			// show IP address, gateway, netmask and DHCP state
	show-net		show_net		// show LwIP network statistics: RX/TX counts, errors
	show-http		show_http		// show HTTP server state and active connections
	show-telnet		show_telnet		// show Telnet server state and active connections
	show-stats      show_stats      // show packet stats
End

Words // usb
	show-usb		show_usb		// show USB connection state and CDC line status
End

Words // accelerometer
	show-acc		show_acc		// show accelerometer type, orientation (pitch/roll) and tap count
	accon			accel_start		// start accerloeromete
	accoff			accel_stop		// stop accelerometer
	accel_init		accel_init		// init accelerometer interface
	accread			accel_regs_read	// ( a n ) read n accelerometer registers starting at a
	accwrite		accel_reg_write	// ( n a ) write n to accelerometer register a
	xyz				xyz_read		// display current accelerometer values
End

Words // ADC
	show-ADC		show_adc		// show all adc inputs
End

Words // DAC
	show-dac		show_dac		// show DAC sine wave output: frequency, samples, timer registers
	dacon			DAC_Sine_Start	// start sine wave on PA4
	dacoff			DAC_Sine_Stop	// stop sine wave on PA4
	dacset			cli_DAC_Sine_SetFreq // ( f ) set sine wave to frequency f in hz from 1 to 1,000,000
End

Words // DMA
	show-dma		show_dma		// show DMA controller usage: active streams, channels, state, error flags
End

Words // system
	show-sys		show_sys		// show system info: clock frequencies and uptime
	show-timer		show_timer		// show delta timer state and UTC tick counter
	reboot			do_reboot		// reboot the device via NVIC system reset
	show-cli		show_cli		// display cli status
	pins			gpio_dump_all	// show all pins and states
End

Words // system	
	tabto cli_tabTo // (n) move ahead to line position n
	key get_key // ( - c ) return character c from key queue or 0
	key? ask_key // ( - f ) return true if there is a key in the keyq
	te print_te // print counter, compare value and list of time event actions with due dates
	telist te_lists // print TE todo and done list of time events
	pa print_actions // print actions in queue
	tn dumpTeaNames // dump out names in action name array
	mstats machineStats // list stats for machines
	0stats zeroMachineTimes // initialize stats to zero
	teakeycosts show_key_costs // show key hashes and clustering of teatimes table
    latency show_latency // show latency stats for time events and actions
	0latency reset_latency // zero the latency stats; sampled every 10s
	testtime test_time // ( s ) test ticks, timeouts and time for s seconds
	tickms ticks_ms // ( tick - ms ) convert tick to milliseconds
	gtt get_tick_time // print tick and time
	start cliStartTime // create a reference point
	end cliEndTime // print time from start
	play play_events // play out events in event queue and restart recording
	stop record_event_off // stop recording events
	record record_events_on // start recording events
	echoon autoEchoOn // turn on key echo
	echooff autoEchoOff // turn off key echo
	nap nap_for // (n) take a nap for n milliseconds
End
